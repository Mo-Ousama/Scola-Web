# Welcome to Python Programming!
import turtle

# Create a turtle named Timmy
timmy = turtle.Turtle()
timmy.shape("turtle")
timmy.color("blue")

# Draw a colorful spiral
for i in range(100):
    timmy.forward(i * 2)
    timmy.right(91)
    if i % 10 == 0:
        timmy.color("red")
    elif i % 10 == 5:
        timmy.color("green")

print("Amazing! You just created art with code!")
